## 2. Blockziel (aus dem Blockplan, woertlich)

> **B2 - Kampf messen (kein Umbau am Spiel):** `tools/mess_kampf.py` - echte Schlachten ueber
> `kampf.fuehre_schlacht`, 40 Laeufe je Wallstufe, Beute ueber `kampfbericht.beute_am_ziel`,
> Reisekosten ueber die vorhandene Wegzeitfunktion.
> **Muss messen:** Trefferquote/Verluste je Wallstufe 0...10, Schwellenverhalten (die Wallpruefung
> rechnet mit **SK**, nicht mit AT - Register 54), Beute-Staffel 10/30/60/100 % **gegen die
> mitnehmbare Menge**, Ladungsdeckel je Karrenart (40/100/250/1.000/2.500, Wagenbau-Bonus +10 %/Stufe,
> Deckel Faktor 2,0).
> **Abnahme:** 40 Laeufe je Stufe sind im Bericht als Tabelle sichtbar; auffaellige Ausreisser werden
> benannt, nicht geglaettet.

def fuehre_schlacht(angreifer: Dict[str, int], verteidiger: Dict[str, int],
                    wall_stufe: int = 0, wall_zustand: float = 100.0,
                    angreifer_karren: Optional[Dict[str, int]] = None,
                    verteidiger_karren: Optional[Dict[str, int]] = None,
                    vorraete: Optional[Dict[str, float]] = None,
                    rng: Optional[random.Random] = None,
                    forschung_angreifer: Optional[dict] = None,
                    forschung_verteidiger: Optional[dict] = None) -> Dict[str, Any]:
    """Führt reproduzierbar eine Schlacht nach der späteren Originalbeschreibung aus.

    Nicht im Dokument geregelte Gerätelager- und Beutemechanik bleibt aus dem
    älteren Handbuch erhalten; siehe GPT-KA01 Quell-/Abweichungsregister.
    """
    if rng is None:
        rng = random.Random()
    stufe = max(0, min(10, int(wall_stufe)))
    zustand = max(0.0, min(100.0, float(wall_zustand)))
    A, V = _einheiten(angreifer, "Angreifer"), _einheiten(verteidiger, "Verteidiger")
    a_wissen = forschung_angreifer or {}
    v_wissen = forschung_verteidiger or {}
    geraet = dict(angreifer_karren or {})
    for n, k in (angreifer or {}).items():
        if E.rolle(n) == "belagerung":
            # Beide Aufrufkonventionen sind erlaubt: Gerät kann im Angreifer-
            # Dict ODER separat übergeben werden. Kein doppeltes Belagerungsgerät.
            geraet.setdefault(n, max(0, int(k)))
    belagerung = {n: k for n, k in geraet.items() if E.rolle(n) == "belagerung"}
    logs = [f"Kampfmodell {STAND}; Wall Stufe {stufe} Zustand {zustand:.1f}%"]
    unsicher = ["Forschungsstand nicht übergeben: Wert 0 statt fiktiver Bot-Forschung"] if (
        forschung_angreifer is None or forschung_verteidiger is None) else []
    result = {"stand": STAND, "ausgang": "unentschieden",
              "kampf_stattgefunden": False, "vorrunde_bestanden": False,
              "sequenzen": [],
              "verluste_angreifer": {}, "verluste_verteidiger": {},
              "geflohen": [], "mauer_zustand": zustand, "beute": {},
              "bericht": "", "unsicher": unsicher}
    def fertig(ausgang):
        result["ausgang"] = ausgang
        result["verluste_angreifer"] = dict(Counter(u["name"] for u in A if u["tot"]))
        result["verluste_verteidiger"] = dict(Counter(u["name"] for u in V if u["tot"]))
        result["geflohen"] = [f"{u['name']} (Angreifer)" for u in A if u["rueckzug"]]
        result["bericht"] = "\n".join(logs)
        return result

    # Ohne Nahkampfeinheiten darf es weder automatischen Sieg noch Beute geben.
    # Ein reiner Fernkampftrupp darf schiessen, erobert aber nichts im Nahkampf.
    if not A:
        logs.append("Keine eigenen Kampfeinheiten: kein Kampf, keine Beute")
        return fertig("unentschieden")

    # Wallvoraussetzung vor der Schlacht; SK und Intervalle bleiben Originaldaten.
    if stufe:
        wall = spieldaten.wallstufen()[stufe - 1]
        # Neu gefundenes Originalforum: die Fußtruppen-Schwelle wird von
        # NAHKAMPFeinheiten überwunden. Fernkämpfer zählen nicht dazu;
        # sonst könnten reine Bogenschützen ohne Sturmtruppen Wall 1 öffnen.
        foot_sk = sum(E.schlagkraft(u["name"]) for u in _nah(A))
        siege_sk = sum(E.schlagkraft(n) * int(k) for n, k in belagerung.items())
        faktor = zustand / 100.0
        if foot_sk < float(wall["schwellenwert_fussttruppen"]) * faktor:
            logs.append("Fußtruppen: Schwelle nicht erreicht; die Truppen ziehen sich zurueck; kein Kampf")
            return fertig("unentschieden")
        schwelle_bel = wall.get("schwellenwert_belagerungswaffen")
        if schwelle_bel is not None and siege_sk < float(schwelle_bel) * faktor:
            logs.append("Belagerungs-Schlagkraft zu gering: Schwelle nicht erreicht; die Truppen ziehen sich zurueck; kein Kampf")
            return fertig("unentschieden")

    result["vorrunde_bestanden"] = True
    result["kampf_stattgefunden"] = True

    # Schutz in Türmen: jeweils beste 10/30 Nahkämpfer des Angreifers.
    schutz = 10 * int(belagerung.get("Belagerungsturm", 0)) + \
             30 * int(belagerung.get("Großer Belagerungsturm", 0))
    geschuetzt = {id(u) for u in sorted(_nah(A),
                 key=lambda u: (u["hi"], u["kon"]), reverse=True)[:max(0, schutz)]}

    # Mauerstufe + 1 echte Vorrunden – selbst ohne Mauer genau eine.
    # Beide Seiten treffen Entscheidungen vor Verrechnung der Rundenerschöpfung.
    for runde in range(stufe + 1):
        schuesse = []
        for schuetze_seite, gegner, wissen, abzug in (
            (A, V, a_wissen, 0.05 * stufe * (zustand / 100.0)),
            (V, A, v_wissen, 0.0),
        ):
            for schuetze in _fern(schuetze_seite):
                ziele = _nah(gegner) or _fern(gegner)
                if not ziele:
                    continue
                ziel = rng.choice(ziele)
                chance = schuetze["fern"] / 100.0 + \
                         0.02 * _forschungsstufe(wissen, "Bogenbau") - abzug
                if schuetze_seite is V and id(ziel) in geschuetzt:
                    chance -= 0.30
                chance = max(0.0, min(1.0, chance))
                if rng.random() < chance:
                    schuesse.append((ziel, schuetze["fern_schaden"]))
        for ziel, schaden in schuesse:
            ziel["erschoepfung"] += schaden
        zeilen = [f"Fernkampf-Vorrunde {runde + 1}/{stufe + 1}: {len(schuesse)} Treffer"]
        for u in A + V:
            _ausgeschieden(u, rng, zeilen)
        logs.extend(zeilen)
        result["sequenzen"].append("\n".join(zeilen))

    # Mauerschaden erst NACH sämtlichen Schussvorrunden: beeinflusst nächsten Kampf.
    mauerschaden = 0
    for name, menge in (belagerung.items() if stufe else ()):
        minimum, maximum = E.mauer_schaden(name)
        for _ in range(max(0, int(menge))):
            if maximum > 0:
                mauerschaden += max(0, rng.randint(minimum, maximum) - stufe)
    result["mauer_zustand"] = max(0.0, zustand - mauerschaden)
    logs.append(f"Mauerschaden nach Fernkampf: {mauerschaden}; Zustand {result['mauer_zustand']:.1f}%")

    ursprung_fern = len(_fern(A))
    rueckzug_je_runde = math.ceil(ursprung_fern * 0.2)
    nahkampf_sequenzen = 0
    for seq in range(1, 11):
        if not _nah(A) or not _alle(V):
            break
        nahkampf_sequenzen = seq
        protokoll = [f"Nahkampf-Sequenz {seq}"]
        # 20 % der ursprünglich vorhandenen angreifenden Schützen ziehen
        # sich je Sequenz zurück, die stärksten zuerst.
        for u in sorted(_fern(A), key=lambda x: (x["fern"], x["kon"]), reverse=True)[:rueckzug_je_runde]:
            u["rueckzug"] = True
            protokoll.append(f"{u['name']} zieht sich zurück")
        for kaempfer in sorted(_nah(A), key=lambda u: (u["hi"], u["kon"])):
            if not _aktiv(kaempfer):
                continue
            ziele = _nah(V) or _fern(V)
            if not ziele:
                break
            ziel = rng.choice(ziele)
            a_wurf = rng.randint(kaempfer["lo"], kaempfer["hi"])
            v_wurf = rng.randint(ziel["lo"], ziel["hi"])
            a_kampf = (a_wurf * (1 + 0.10 * _forschungsstufe(a_wissen, "Schmiedekunst"))
                       - ziel["ruestung"] * (1 + 0.10 * _forschungsstufe(v_wissen, "Schmiedekunst"))
                       - kaempfer["erschoepfung"])
            v_kampf = (v_wurf * (1 + 0.10 * _forschungsstufe(v_wissen, "Schmiedekunst"))
                       - kaempfer["ruestung"] * (1 + 0.10 * _forschungsstufe(a_wissen, "Schmiedekunst"))
                       - ziel["erschoepfung"])
            protokoll.append(f"{kaempfer['name']} {a_kampf:.2f} gegen {ziel['name']} {v_kampf:.2f}")
            loser = [ziel] if a_kampf > v_kampf else ([kaempfer] if a_kampf < v_kampf
                                                       else [kaempfer, ziel])
            for u in loser:
                u["erschoepfung"] = max(u["erschoepfung"], u["schwelle"])
                _ausgeschieden(u, rng, protokoll)
            if len(loser) == 1:
                sieger = kaempfer if loser[0] is ziel else ziel
                if _aktiv(sieger):
                    sieger["erschoepfung"] += 2
                    _ausgeschieden(sieger, rng, protokoll)
        result["sequenzen"].append("\n".join(protokoll))
        logs.extend(protokoll)
        if not _alle(V) or not _nah(A):
            break

    if not _alle(V) and not _nah(A):
        # Kein Sieger, wenn beide Nahkampfseiten gleichzeitig ausscheiden.
        ausgang = "unentschieden"
    elif not _alle(V) and _alle(A):
        ausgang = "angreifer"
    elif not _nah(A):
        ausgang = "verteidiger"
    else:
        ausgang = "unentschieden"
    # Nicht zurückgezogene Schützen werden bei Niederlage überrannt und
    # müssen wie im Forumstext die Konstitutionsprobe bestehen.
    if ausgang == "verteidiger":
        for u in _fern(A):
            u["inaktiv"] = True
            u["tot"] = rng.randint(1, 100) > u["kon"]
            logs.append(f"{u['name']} überrannt: {'gefallen' if u['tot'] else 'entkommen'}")

    if ausgang == "angreifer":
        result["beute"] = _beute(geraet, vorraete)
        ent, verl = _sieggeraete(verteidiger_karren, nahkampf_sequenzen, rng)
        if verteidiger_karren:
            result["geraete_erhalten_verteidiger"] = ent
            result["geraete_verloren_verteidiger"] = verl
    elif ausgang == "verteidiger":
        ent, verl = _sieggeraete(geraet, nahkampf_sequenzen, rng)
        if geraet:
            result["geraete_erhalten"] = ent
            result["geraete_verloren"] = verl
    logs.append(f"Ausgang: {ausgang}; Nahkampf-Sequenzen: {nahkampf_sequenzen}")
    return fertig(ausgang)

def vorrunde(angreifer_fussttruppen_sk: float,
             angreifer_belagerung_sk: float,
             wall_stufe: int,
             wall_zustand_prozent: float = 100.0) -> Dict[str, Any]:
    """Prueft die Vorrunde nach Handbuch Seite 20/21.

    `wall_zustand_prozent` ist optional und stand nicht im geforderten Aufruf; ohne Angabe
    gilt der volle Zustand (100 %). Eine beschaedigte Mauer verlangt laut Handbuch
    "prozentual weniger" Schlagkraft - der Faktor ist Zustand/100.

    Rueckgabe:
      {'bestanden': bool, 'schwelle': {...}, 'grund': text}
      schwelle = {
        'fusstruppen':        {'nennwert', 'erforderlich', 'vorhanden', 'erreicht'},
        'belagerungswaffen':  {'nennwert', 'erforderlich', 'vorhanden', 'erreicht'|None},
        'wall_stufe', 'wall_name', 'wall_zustand_prozent', 'faktor',
      }
    """
    stufe = int(wall_stufe)
    zustand = _als_zahl(wall_zustand_prozent, 100.0)
    faktor = max(0.0, zustand) / 100.0

### Funktion `beute_staffel` aus `src/kampfbericht.py` (Zeilen 227-277)
```python
def beute_staffel(einheiten_des_siegers: Dict[str, int],
                  vorraete: Dict[str, float],
                  wall_zustand_vor: Any = None,
                  verteidiger_uebrig: Any = None,
                  forschung: Any = None) -> Dict[str, int]:
    """Beute nach Daniels Staffelung (Registerpunkt 45).

    Anteil = BEUTE_STAFFEL_PROZENT[beute_lage(...)], zusaetzlich begrenzt durch
      * die Gesamtladung der mitgefuehrten Karren und
      * den tatsaechlich vorhandenen Vorrat je Rohstoff.
    Die Verteilung auf die Rohstoffe ist anteilig nach Vorratsmenge (das Handbuch
    regelt die Aufteilung nicht).

    Beispiele (Vorschlagswerte): Wall heil, 1.000 je Rohstoff, Handkarren (Ladung 40)
    -> 10 Prozent = 100, aber nur 40 passen hinein. Ohne Verteidiger -> 100 Prozent,
    begrenzt durch Ladung. Kein Karren -> keine Beute (belegt).
    """
    kapazitaet = E.ladung_der_gruppe(einheiten_des_siegers, forschung)
    if kapazitaet <= 0:
        return {}

    vorhanden: Dict[str, int] = hilfen.positive_ganze_werte(vorraete)
    gesamt = sum(vorhanden.values())
    if gesamt <= 0:
        return {}

    anteil = BEUTE_STAFFEL_PROZENT[beute_lage(wall_zustand_vor, verteidiger_uebrig)] / 100.0
    # Anteilig heisst: gegen die WIRKLICH mitnehmbare Menge rechnen, nicht gegen die ideale.
    # Bisher stand hier `int(anteil_stoff * ziel_menge)` und danach ein Deckel `restplatz` -
    # dadurch frass der alphabetisch erste Rohstoff die ganze Ladung (Befund 31.10.2026 aus
    # `test_kampf_einhaengung`: erwartet Holz 50 / Stein 30, geliefert Holz 80 / Stein 0).
    ziel = min(kapazitaet, int(gesamt * anteil))
    if ziel <= 0:
        return {}
    genommen: Dict[str, int] = {}
    for stoff in sorted(vorhanden):
        nimm = min(int(vorhanden[stoff] / gesamt * ziel), vorhanden[stoff])
        if nimm > 0:
            genommen[stoff] = nimm
    # Rest, der durch das Abrunden liegen bleibt, nach dem groessten Bestand auffuellen -
    # nie ueber den vorhandenen Vorrat hinaus. So ist die Summe exakt und die Verteilung fair.
    rest = ziel - sum(genommen.values())
    for stoff in sorted(vorhanden, key=lambda s: (-vorhanden[s], s)):
        if rest <= 0:
            break
        platz = vorhanden[stoff] - genommen.get(stoff, 0)
        nimm = min(rest, platz)
        if nimm > 0:
            genommen[stoff] = genommen.get(stoff, 0) + nimm
            rest -= nimm
    return genommen
```

                  vorraete: Dict[str, float],
                  ausgang: Any = "angreifer",
                  wall_zustand_vor: Any = None,

    """Beute berechnen UND beim Ziel abbuchen - Rueckgabe (beute, vorraete_danach).

    Warum es diese Funktion gibt (Befund der externen Gegenpruefung, 31.10.2026, Nr. 6):
    Der Auftrags-Pfad schrieb die Beute dem Spieler gut, ohne sie beim Ziel abzuziehen -
    Gueter aus dem Nichts. Der Bot-Pfad zog ab, rechnete die Menge aber mit einer anderen
    Formel (`kampf._beute`: nur Ladungsgrenze). Jetzt gilt an EINER Stelle:

      * Beute nur bei Angreifersieg (belegt: Handbuch S. 23, Z. 1307-1315),
      * der Anteil kommt aus `beute_staffel` (Wallzustand VOR dem Kampf, ueberlebende
        Verteidiger, Ladung, vorhandener Vorrat),
      * was gutgeschrieben wird, wird beim Ziel abgebucht; kein Bestand wird negativ.
    """
    lager: Dict[str, float] = {}
    for stoff, menge in (vorraete or {}).items():
        wert = _als_zahl(menge)
        lager[str(stoff)] = max(0.0, wert)

    if str(ausgang or "") != "angreifer":
        return {}, lager

    beute_werte = dict(beute_staffel(einheiten_des_siegers, lager,
                                    wall_zustand_vor=wall_zustand_vor,
                                    verteidiger_uebrig=verteidiger_uebrig,
                                    forschung=forschung))
    for stoff, menge in beute_werte.items():
        lager[stoff] = round(max(0.0, lager.get(stoff, 0.0) - float(menge)), 2)
    return beute_werte, lager

def mauer_schaden(belagerung: Dict[str, int],
                  wall_stufe: int,
                  wall_zustand_prozent: float,
                  rng: Optional[random.Random] = None) -> Dict[str, Any]:
    """Mauerschaden nach Handbuch Seite 21/22.

    Je mitgefuehrter Belagerungswaffe wird EINZELN gewuerfelt: Zufallswert im Schadensbereich
    der Waffe minus Ausbaustufe der Mauer; ein positiver Rest geht vom Zustand ab. Der
    Zustand faellt nie unter 0 %.

    Rueckgabe:
      {'mauer_zustand_prozent', 'vorher_prozent', 'schaden_gesamt',
       'wall_stufe', 'je_waffe': {name: {'anzahl', 'wuerfe', 'rest', 'schaden'}}}
    """
    rng = rng or random.Random()
    stufe = int(wall_stufe)
    vorher = max(0.0, _als_zahl(wall_zustand_prozent, 100.0))

    je_waffe: Dict[str, Dict[str, Any]] = {}
    gesamt = 0
    for name, anzahl in (belagerung or {}).items():
        n = max(0, _als_int(anzahl))
        lo, hi = E.mauer_schaden(name)
        wuerfe, rest = [], []
        for _ in range(n):
            wurf = rng.randint(int(lo), int(hi))
            r = max(0, int(wurf) - stufe)
            wuerfe.append(int(wurf))
            rest.append(int(r))
            gesamt += int(r)
        if name not in je_waffe:
            je_waffe[name] = {"anzahl": n, "wuerfe": wuerfe, "rest": rest,
                              "schaden": sum(rest)}

    neu = max(0.0, vorher - gesamt)
    return {
        "mauer_zustand_prozent": neu,
        "vorher_prozent": vorher,
        "schaden_gesamt": gesamt,
        "wall_stufe": stufe,
        "je_waffe": je_waffe,
    }

## 8. Was das Werkzeug messen MUSS

1. **Wallstufen 0 bis 10, je 40 Laeufe** mit derselben Angreiferaufstellung und derselben
   Verteidigeraufstellung: Tabelle mit Stufe, Anteil bestandener Vorrunden, Anteil Siege,
   Verluste Angreifer (Min/Median/Max), Verluste Verteidiger, ob der Wall ueberwunden wurde.
2. **Schwellenverhalten:** ab welcher Summe Schlagkraft die Vorrunde besteht, je Stufe. Vergleiche
   die gemessene Schwelle mit der Schwelle, die `kampfbericht._schwelle_wall` bzw. die
   Wallstufen-Tabelle in `data/buildings.json` nennt. Nenne jede Abweichung.
3. **Beute:** die Staffel 10 / 30 / 60 / 100 Prozent gegen die **mitnehmbare Menge** (Karrenladung)
   bei kleinem und grossem Vorrat (z. B. 100 und 10.000 je Rohstoff). Belegt werden muss, dass
   zuerst die Ladung deckelt und der Anteil danach - nicht umgekehrt.
4. **Ladung:** je Karrenart der Deckel (40/100/250/1.000/2.500), mit Wagenbau 1, 5, 10 und ueber
   der Hoechststufe; Deckel Faktor 2,0.
5. **Reisekosten:** Gold, Nahrung und Mana je Feld ueber die vorhandene Wegzeit-/Kostenfunktion,
   fuer 10, 40 und 100 Einheiten und fuer Entfernungen 1, 5, 20 Felder; Kennzeichnung, was belegt
   und was VORSCHLAG ist.
6. **Gegenprobe zum AT/SK-Fehler:** dieselbe Aufstellung einmal mit `E.angriff` (AT) und einmal mit
   `E.schlagkraft` (SK) durch die Vorrunde schicken und beide Ergebnisse nebeneinander zeigen.
   Das ist der Beweis, dass Register 54 wirklich behoben ist.
