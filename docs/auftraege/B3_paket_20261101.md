def wirkung_anwenden(bot: dict, entscheidung: dict, jetzt: float) -> dict:
    """Schreibt eine Entscheidung in einen NEUEN Bot fort (Eingabe unveraendert,
    tiefe Kopie): naechster Zustand, Marsch-/Flucht-Zeitstempel, letzter Abstand
    und das Gedaechtnis. Nach einer Niederlage wird der Bot aggressiver."""
    neu = copy.deepcopy(bot) if isinstance(bot, dict) else {}
    e = entscheidung if isinstance(entscheidung, dict) else {}
    w = e.get("wirkung") if isinstance(e.get("wirkung"), dict) else {}
    jetzt = _zahl(jetzt, 0.0)

    ged = dict(neu.get("gedaechtnis")) if isinstance(neu.get("gedaechtnis"), dict) else {}
    for schluessel, wert in GEDAECHTNIS_DEFAULT.items():
        ged.setdefault(schluessel, wert)

    naechster = w.get("naechster_zustand") or e.get("naechster_zustand")
    if naechster in ZUSTAENDE:
        neu["zustand"] = naechster

    if "marsch_ankunft" in w:
        ankunft = _zahl(w["marsch_ankunft"], 0.0)
        neu["marsch_ankunft"] = ankunft
        if ankunft > jetzt:
            neu["naechster_marsch_h"] = round((ankunft - jetzt) / 3600.0, 1)
    if "flucht_ende" in w:
        neu["flucht_ende"] = _zahl(w["flucht_ende"], 0.0)
    if e.get("abstand") is not None:
        neu["letzter_abstand"] = _ganzzahl(e["abstand"], 0)

    schlaeger = e.get("geschlagen_von") or w.get("geschlagen_von")
    if schlaeger:
        ged["geschlagen_von"] = str(schlaeger)
        delta = _zahl(w.get("aggression_delta"), AGGRESSION_ZUWACHS_BEI_NIEDERLAGE)
        neu["aggression"] = round(min(1.0, _klemme(neu.get("aggression", 0.0)) + delta), 2)

    beute_weg = e.get("beute_verloren") or w.get("beute_verloren")
    if beute_weg is not None:
        ged["beute_verloren"] = {"menge": _ganzzahl(beute_weg, 0), "zeit": jetzt}
    staerke = e.get("staerke_zuletzt") or w.get("staerke_zuletzt")
    if staerke is not None:
        ged["staerke_zuletzt"] = {"punkte": _ganzzahl(staerke, 0), "zeit": jetzt}
    if e.get("aktion") in ("raeubern", "raechen", "fliehen", "marschieren", "suchen"):
        ged["letzter_kontakt"] = jetzt

    neu["gedaechtnis"] = ged
    return neu

    BELEGT (Handbuch 2009, Seiten 4-6; Registerpunkte 31/32): Kriegspunkte = Summe der
    Einheitenwerte + Wallstufe hoch drei; Forschungspunkte = Summe der Forschungsstufen hoch
    drei; Gesamtpunkte = (Krieg + Forschung) / 2, abgerundet.

    Warum Gesamtpunkte und nicht Kriegspunkte (Entscheidung Daniel 31.10.2026, Frage 10):
    die Ruhmeshalle und das Goetterschutz-Band vergleichen mit `punktestand()['gesamt']` des
    Spielers. Solange der Bot nur Kriegspunkte fuehrte, wurde Krieg gegen Gesamt verglichen -
    der Bot erschien doppelt so stark, wie er ist.

    Forschung des Bots: hat er `forschung_stufen` (Stufe je Linie), gilt die belegte Formel
    direkt. Sonst zaehlt sein Zaehler `geforscht` als EINE Forschungslinie - VORSCHLAG, weil
    das Handbuch die Bot-Forschung nicht regelt (Bots haben im Remake keine Forschungsstaende).

    Vorher war `punkte` ein WUERFELWERT (`rng.randint(0, 300)`): Rangliste, Ruhmeshalle und
    Schutzband waren Zufall. Diese Funktion ist die EINE Stelle, an der Bot-Punkte entstehen;
    `wirkung_anwenden` schreibt sie nach jeder Aenderung fort.
    """
    bot = bot or {}
    armee = bot.get("armee") if isinstance(bot.get("armee"), dict) else {}
    wall = bot.get("wall_stufe", 0)
    stufen = bot.get("forschung_stufen")
    if not isinstance(stufen, dict):
        versuche = bot.get("geforscht", 0)
        try:
            versuche = int(versuche)
        except (TypeError, ValueError):
            versuche = 0
        stufen = {"Forschung": versuche} if versuche > 0 else {}
    try:
        krieg = int(punkte.kriegspunkte(armee, wall))
        forschung = int(punkte.forschungspunkte(stufen))
    except (TypeError, ValueError):
        return 0
    return (krieg + forschung) // 2

1. **Anbindung**: `bots_fortschreiben` fragt je Bot `bottaktik.tick` und wendet die Entscheidung mit

5. **Demo-Regel**: kein Bot-Angriff; das Verhalten muss auch ohne Angriff sichtbar sein.
